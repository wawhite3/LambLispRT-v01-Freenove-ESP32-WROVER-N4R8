716 LL_String() { ptr = 0; set(""); }

732 LL_String concat(const char *s1, const char *s2) {
733 int n1 = strlen(s1);
734 int n2 = strlen(s2);
735 char *s3 = new char[n1 + n2 + 1];
736
737 char *p = s3;
738 while (n1--) *p++ = *s1++;
739 while (n2--) *p++ = *s2++;
740 *p = '\0';
741 return s3;
742 }
743
744 LL_String &append(const char *s1) {
745 const char *s0 = ptr;
746 int n0 = s0 ? strlen(s0) : 0;
747 int n1 = strlen(s1);
748 char *s2 = new char[n0 + n1 + 1];
749
750 char *p = s2;
751 while (n0--) *p++ = *s0++;
752 while (n1--) *p++ = *s1++;
753 *p = '\0';
754
755 if (ptr) { delete[] ptr; ptr = 0; }
756 ptr = s2;
757 return *this;
758 }

1070inline int ll_u8_seqlen(unsigned char b0)
1071{
1072 if (b0 < 0x80) return 1;
1073 if ((b0 & 0xE0) == 0xC0) return 2;
1074 if ((b0 & 0xF0) == 0xE0) return 3;
1075 if ((b0 & 0xF8) == 0xF0) return 4;
1076 return 1;
1077}
1078
1081inline ll_codepoint_t ll_u8_decode(const char *p, int *nb)
1082{
1083 unsigned char b0 = (unsigned char) p[0];
1084 int n = ll_u8_seqlen(b0);
1085 if (nb) *nb = n;
1086 if (n == 1) return (ll_codepoint_t) b0;
1087 ll_codepoint_t cp = b0 & (0xFF >> (n + 1));
1088 for (int i = 1; i < n; i++) {
1089 unsigned char c = (unsigned char) p[i];
1090 if ((c & 0xC0) != 0x80) { if (nb) *nb = 1; return (ll_codepoint_t) b0; }
1091 cp = (cp << 6) | (c & 0x3F);
1092 }
1093 return cp;
1094}
1095
1097inline int ll_u8_encode(ll_codepoint_t cp, char *out)
1098{
1099 if (cp < 0x80) { out[0] = (char) cp; return 1; }
1100 if (cp < 0x800) { out[0] = (char)(0xC0 | (cp >> 6)); out[1] = (char)(0x80 | (cp & 0x3F)); return 2; }
1101 if (cp < 0x10000) { out[0] = (char)(0xE0 | (cp >> 12)); out[1] = (char)(0x80 | ((cp >> 6) & 0x3F));
1102 out[2] = (char)(0x80 | (cp & 0x3F)); return 3; }
1103 out[0] = (char)(0xF0 | (cp >> 18)); out[1] = (char)(0x80 | ((cp >> 12) & 0x3F));
1104 out[2] = (char)(0x80 | ((cp >> 6) & 0x3F)); out[3] = (char)(0x80 | (cp & 0x3F)); return 4;
1105}
